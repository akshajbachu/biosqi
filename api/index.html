    def analyze(
    recording: Recording,
    *,
    window_seconds: float | None = None,
    step_seconds: float | None = None,
    threshold: float = 0.5,
    model: FusionModel | None = None,
    include_partial: bool = True,
) -> AnalysisResult:
    """Analyze every channel and aggregate each window conservatively by median.

    Scores are calibrated estimates from the selected model, not guarantees of
    clinical usability. The threshold is an application policy and should be
    validated for the intended downstream task.
    """

    if not 0.0 <= threshold <= 1.0:
        raise ValueError("threshold must lie in [0, 1]")
    profile: SignalProfile = get_profile(recording.signal_type)
    window_length = profile.window_seconds if window_seconds is None else window_seconds
    step_length = profile.step_seconds if step_seconds is None else step_seconds
    fusion = load_default_model() if model is None else model
    warnings: list[str] = []
    window_results: list[WindowResult] = []
    for window in iter_windows(
        recording,
        window_length,
        step_length,
        include_partial=include_partial,
    ):
        channels: list[ChannelResult] = []
        for name, values in zip(recording.channel_names, window.data, strict=True):
            indicators = compute_indicators(values, recording.sample_rate, profile)
            invalid_count = sum(not indicator.valid for indicator in indicators)
            if invalid_count:
                warnings.append(
                    f"window {window.index}, channel {name}: "
                    f"{invalid_count} indicator(s) unavailable"
                )
            prediction = fusion.predict(indicators)
            channels.append(
                ChannelResult(
                    channel=name,
                    score=prediction.raw_score,
                    calibrated_score=prediction.calibrated_score,
                    indicators=indicators,
                    artifacts=attribute_artifacts(indicators, fusion),
                )
            )
        channel_tuple = tuple(channels)
        raw_score = float(np.median([channel.score for channel in channel_tuple]))
        calibrated_score = float(np.median([channel.calibrated_score for channel in channel_tuple]))
        window_results.append(
            WindowResult(
                index=window.index,
                start=window.start,
                stop=window.stop,
                start_seconds=window.start_seconds,
                stop_seconds=window.stop_seconds,
                score=raw_score,
                calibrated_score=calibrated_score,
                usable=calibrated_score >= threshold,
                channels=channel_tuple,
                artifacts=_aggregate_artifacts(channel_tuple),
            )
        )
    if not window_results:
        warnings.append("recording contains no samples")
    return AnalysisResult(
        source=recording.source,
        signal_type=recording.signal_type,
        sample_rate=recording.sample_rate,
        channel_names=recording.channel_names,
        n_samples=recording.n_samples,
        window_seconds=window_length,
        step_seconds=step_length,
        threshold=threshold,
        windows=tuple(window_results),
        model_name=fusion.name,
        warnings=tuple(dict.fromkeys(warnings)),
    )

    def read_csv(
    path: str | Path,
    *,
    sample_rate: float,
    signal_type: SignalType = SignalType.GENERIC,
) -> Recording:
    """Read a numeric CSV/TSV where rows are samples and columns are channels."""

    source = Path(path)
    with source.open("r", encoding="utf-8-sig", newline="") as handle:
        sample = handle.read(8192)
        handle.seek(0)
        try:
            dialect = csv.Sniffer().sniff(sample, delimiters=",;\t")
        except csv.Error:
            dialect = csv.excel
        rows = list(csv.reader(handle, dialect))
    rows = [row for row in rows if row and any(cell.strip() for cell in row)]
    if not rows:
        raise ValueError("CSV file is empty")

    def numeric(row: list[str]) -> bool:
        try:
            [float(cell.strip()) for cell in row]
        except ValueError:
            return False
        return True

    has_header = not numeric(rows[0])
    names = tuple(cell.strip() or f"ch{i}" for i, cell in enumerate(rows[0])) if has_header else ()
    body = rows[1:] if has_header else rows
    width = len(body[0]) if body else 0
    if width == 0 or any(len(row) != width for row in body):
        raise ValueError("CSV rows must have a consistent, non-zero column count")
    if names and len(names) != width:
        raise ValueError("CSV header and data column counts differ")
    try:
        data = np.asarray([[float(cell.strip()) for cell in row] for row in body], dtype=np.float64)
    except ValueError as exc:
        raise ValueError("CSV contains non-numeric data") from exc
    return Recording(data.T, sample_rate, names, signal_type, str(source))

    def read_edf(
    path: str | Path,
    *,
    signal_type: SignalType = SignalType.GENERIC,
) -> Recording:
    """Read EDF/BDF via ``pyedflib``; channels must share a sample rate."""

    pyedflib = _optional_import("pyedflib", "data")
    source = Path(path)
    reader = pyedflib.EdfReader(str(source))
    try:
        rates = np.asarray(reader.getSampleFrequencies(), dtype=np.float64)
        if rates.size == 0:
            raise ValueError("EDF file contains no channels")
        if not np.allclose(rates, rates[0]):
            raise ValueError(
                "EDF channels have different sample rates; select or resample channels first"
            )
        data = np.vstack([reader.readSignal(index) for index in range(reader.signals_in_file)])
        names = tuple(str(name).strip() for name in reader.getSignalLabels())
        metadata = {"physical_dimensions": list(reader.getPhysicalDimensions())}
    finally:
        reader.close()
    return Recording(data, float(rates[0]), names, signal_type, str(source), metadata)

    def read_numpy(
    path: str | Path,
    *,
    sample_rate: float | None = None,
    signal_type: SignalType = SignalType.GENERIC,
    channel_axis: int = 0,
) -> Recording:
    """Read ``.npy`` or ``.npz`` arrays.

    An NPZ may store ``data``, ``sample_rate`` (or ``fs``), and ``channel_names``.
    Plain NPY inputs require ``sample_rate``.
    """

    source = Path(path)
    loaded = np.load(source, allow_pickle=False)
    names: tuple[str, ...] = ()
    metadata: dict[str, Any] = {}
    if isinstance(loaded, np.lib.npyio.NpzFile):
        try:
            keys = loaded.files
            data_key = "data" if "data" in keys else keys[0] if keys else None
            if data_key is None:
                raise ValueError("NPZ archive contains no arrays")
            data = np.asarray(loaded[data_key])
            if sample_rate is None:
                for key in ("sample_rate", "fs"):
                    if key in keys:
                        sample_rate = float(np.asarray(loaded[key]).reshape(-1)[0])
                        break
            if "channel_names" in keys:
                names = tuple(str(name) for name in np.asarray(loaded["channel_names"]).tolist())
            metadata["npz_key"] = data_key
        finally:
            loaded.close()
    else:
        data = np.asarray(loaded)
    if sample_rate is None:
        raise ValueError("sample_rate is required unless stored in an NPZ archive")
    oriented = _orient(data, channel_axis)
    return Recording(oriented, sample_rate, names, signal_type, str(source), metadata)

    def read_recording(
    path: str | Path,
    *,
    sample_rate: float | None = None,
    signal_type: SignalType = SignalType.GENERIC,
    channel_axis: int = 0,
) -> Recording:
    """Dispatch to a reader based on the source suffix."""

    source = Path(path)
    suffix = source.suffix.lower()
    if suffix in {".npy", ".npz"}:
        return read_numpy(
            source,
            sample_rate=sample_rate,
            signal_type=signal_type,
            channel_axis=channel_axis,
        )
    if suffix in {".csv", ".tsv", ".txt"}:
        if sample_rate is None:
            raise ValueError("sample_rate is required for delimited text files")
        return read_csv(source, sample_rate=sample_rate, signal_type=signal_type)
    if suffix in {".edf", ".bdf"}:
        return read_edf(source, signal_type=signal_type)
    if suffix in {".hea", ".dat"} or not suffix:
        return read_wfdb(source, signal_type=signal_type)
    raise ValueError(f"unsupported input format: {suffix or '<none>'}")

    def read_wfdb(
    path: str | Path,
    *,
    signal_type: SignalType = SignalType.ECG,
) -> Recording:
    """Read a WFDB record using the optional ``wfdb`` package."""

    wfdb = _optional_import("wfdb", "data")
    source = Path(path)
    record_name = str(source.with_suffix("")) if source.suffix in {".hea", ".dat"} else str(source)
    record = wfdb.rdrecord(record_name)
    data = np.asarray(record.p_signal, dtype=np.float64).T
    names = tuple(str(name) for name in getattr(record, "sig_name", ()))
    metadata = {"units": list(getattr(record, "units", ()))}
    return Recording(data, float(record.fs), names, signal_type, str(source), metadata)

    def get_profile(signal_type: SignalType | str) -> SignalProfile:
    """Return the profile for an enum or case-insensitive string."""

    try:
        kind = (
            signal_type if isinstance(signal_type, SignalType) else SignalType(signal_type.lower())
        )
    except ValueError as exc:
        supported = ", ".join(item.value for item in SignalType)
        raise ValueError(
            f"unsupported signal type {signal_type!r}; choose from {supported}"
        ) from exc
    return PROFILES[kind]

            stop=stop,

    def seconds_to_samples(seconds: float, sample_rate: float, *, name: str = "duration") -> int:
    """Convert seconds to samples using stable half-up rounding."""

    if not np.isfinite(seconds) or seconds <= 0:
        raise ValueError(f"{name} must be finite and positive")
    if not np.isfinite(sample_rate) or sample_rate <= 0:
        raise ValueError("sample_rate must be finite and positive")
    samples = int(np.floor(seconds * sample_rate + 0.5))
    if samples < 1:
        raise ValueError(f"{name} is shorter than one sample")
    return samples

    def window_bounds(
    n_samples: int,
    sample_rate: float,
    window_seconds: float,
    step_seconds: float,
    *,
    include_partial: bool = True,
) -> tuple[tuple[int, int], ...]:
    """Return deterministic half-open bounds, without padding or duplicated tails."""

    if n_samples < 0:
        raise ValueError("n_samples cannot be negative")
    size = seconds_to_samples(window_seconds, sample_rate, name="window_seconds")
    step = seconds_to_samples(step_seconds, sample_rate, name="step_seconds")
    if n_samples == 0:
        return ()
    if n_samples <= size:
        return ((0, n_samples),) if include_partial or n_samples == size else ()

    starts = list(range(0, n_samples - size + 1, step))
    bounds = [(start, start + size) for start in starts]
    next_start = (starts[-1] + step) if starts else 0
    if include_partial and bounds[-1][1] < n_samples and next_start < n_samples:
        bounds.append((next_start, n_samples))
    return tuple(bounds)

    def baseline_wander_ratio(
    values: npt.ArrayLike,
    sample_rate: float,
    baseline_band: tuple[float, float],
    physiological_band: tuple[float, float],
) -> IndicatorResult:
    """Measure baseline-band power relative to physiological-band power.

    Baseline power ratios are motivated by Clifford et al., Physiol. Meas.
    33(9), 2012. Profile bands and the ratio-to-score transform are heuristic.
    """

    x, finite_fraction = _prepare(values)
    if x is None:
        return _invalid("baseline_wander_ratio", "baseline_wander", "too few finite samples")
    frequencies, power = _welch(x, sample_rate)
    baseline = _band_power(frequencies, power, baseline_band)
    physiological = _band_power(frequencies, power, physiological_band)
    ratio = baseline / max(baseline + physiological, _EPS)
    return IndicatorResult(
        "baseline_wander_ratio",
        ratio,
        float(np.clip((1.0 - ratio) * finite_fraction, 0.0, 1.0)),
        "baseline_wander",
        {"baseline_power": baseline, "physiological_power": physiological},
    )

    def beat_agreement_template(
    values: npt.ArrayLike,
    sample_rate: float,
    signal_type: SignalType,
    rate_bpm: tuple[float, float] | None,
) -> IndicatorResult:
    """Compare two independent beat detectors and beat-template morphology.

    Detector agreement as a quality concept follows Clifford et al., Physiol.
    Meas. 33(9), 2012; template correlation follows Orphanidou et al., IEEE
    J. Biomed. Health Inform. 19(3), 2015. The 150 ms tolerance and equal score
    weighting are explicit heuristics, not clinical thresholds.
    """

    if signal_type not in {SignalType.ECG, SignalType.PPG} or rate_bpm is None:
        return _invalid("beat_agreement_template", "beat_inconsistency", "not applicable")
    x, finite_fraction = _prepare(values, minimum=max(16, int(sample_rate)))
    if x is None:
        return _invalid("beat_agreement_template", "beat_inconsistency", "window is too short")
    first = detect_peaks_prominence(x, sample_rate, signal_type, rate_bpm)
    second = detect_peaks_energy(x, sample_rate, signal_type, rate_bpm)
    tolerance = max(1, round(0.15 * sample_rate))
    agreement = _beat_agreement(first, second, tolerance)
    template = _template_correlation(x, first, max(1, round(0.2 * sample_rate)))
    template_quality = max(0.0, template)
    quality = finite_fraction * (0.6 * agreement + 0.4 * template_quality)
    return IndicatorResult(
        "beat_agreement_template",
        agreement,
        float(np.clip(quality, 0.0, 1.0)),
        "beat_inconsistency",
        {
            "prominence_peak_count": int(first.size),
            "energy_peak_count": int(second.size),
            "agreement_f1": agreement,
            "template_correlation": template,
            "tolerance_ms": 1000.0 * tolerance / sample_rate,
        },
    )

    """Measure clipping, flat-line occupancy, and RMS amplitude.

    Flat-line and saturation checks are common in Li et al., Physiol. Meas.
    29(1), 2008. Without ADC limits, clipping is inferred from repeated extrema;
    the 1% tolerance and score weights are dataset-agnostic heuristics.
    """

    x, finite_fraction = _prepare(values)
    if x is None:
        return _invalid("clipping_flatline_rms", "amplitude_failure", "too few finite samples")
    span = float(np.ptp(x))
    scale = max(span, float(np.nanstd(x)), _EPS)
    tolerance = max(scale * 1e-6, _EPS)
    at_minimum = np.abs(x - np.min(x)) <= tolerance
    at_maximum = np.abs(x - np.max(x)) <= tolerance
    clipping = float(np.mean(at_minimum | at_maximum))
    differences = np.abs(np.diff(x))
    flatline = float(np.mean(differences <= max(scale * 1e-5, _EPS)))
    rms = float(np.sqrt(np.mean(np.square(x - np.mean(x)))))
    dynamic = 0.0 if span <= _EPS else 1.0
    quality = dynamic * finite_fraction * np.exp(-8.0 * clipping - 12.0 * flatline)
    return IndicatorResult(
        "clipping_flatline_rms",
        rms,
        float(np.clip(quality, 0.0, 1.0)),
        "amplitude_failure",
        {"clipping_fraction": clipping, "flatline_fraction": flatline, "rms": rms},
    )

        local = np.abs(filtered[start:stop])
        refined.append(start + int(np.argmax(local)))
    return np.asarray(sorted(set(refined)), dtype=np.int64)

    def distribution_shape(values: npt.ArrayLike) -> IndicatorResult:
    """Return skewness and Pearson kurtosis with a conservative morphology score.

    Orphanidou et al., IEEE J. Biomed. Health Inform. 19(3), 2015 discuss

    the score only down-weights very extreme shape statistics.

    if x is None:
        return _invalid("distribution_shape", "morphology_outlier", "too few finite samples")

    kurtosis = float(stats.kurtosis(x, fisher=False, bias=False))

    quality = np.exp(-penalty) * finite_fraction
    return IndicatorResult(
        "distribution_shape",
        kurtosis,
        float(np.clip(quality, 0.0, 1.0)),
        "morphology_outlier",
        {"skewness": skewness, "pearson_kurtosis": kurtosis},
    )

    def line_noise_ratio(values: npt.ArrayLike, sample_rate: float) -> IndicatorResult:
    """Measure 50/60 Hz power relative to nearby high-frequency power.

    Power-line interference assessment is consistent with Clifford et al.,
    Physiol. Meas. 33(9), 2012. The 1 Hz bands and exponential score mapping
    are transparent engineering choices, not validated clinical thresholds.

        ratios[frequency] = tone / max(neighborhood - tone, _EPS)

    quality = float(np.exp(-2.0 * ratio) * finite_fraction)
    return IndicatorResult(
        "line_noise_ratio",
        ratio,
        float(np.clip(quality, 0.0, 1.0)),
        "line_noise",
        {"dominant_mains_hz": mains, "ratios": str(ratios)},

    def periodicity_autocorrelation(
    values: npt.ArrayLike,
    sample_rate: float,
    rate_bpm: tuple[float, float] | None,
) -> IndicatorResult:
    """Find the strongest normalized autocorrelation at plausible beat lags.

    Autocorrelation SQIs follow the periodicity principle in Li et al.,
    Physiol. Meas. 29(1), 2008. Plausible rate bounds come from the selected
    profile and are broad quality-analysis assumptions, not diagnostic limits.
    """

    x, finite_fraction = _prepare(values, minimum=16)
    if x is None:
        return _invalid("periodicity_autocorrelation", "aperiodicity", "too few finite samples")
    x = np.asarray(signal.detrend(x), dtype=np.float64)
    variance = float(np.dot(x, x))
    if variance <= _EPS:
        return _invalid("periodicity_autocorrelation", "aperiodicity", "constant signal")
    correlation = signal.correlate(x, x, mode="full", method="fft")[x.size - 1 :] / variance
    if rate_bpm is None:
        minimum_lag = max(1, round(0.25 * sample_rate))
        maximum_lag = min(x.size - 1, round(2.0 * sample_rate))
    else:
        minimum_lag = max(1, int(np.floor(60.0 * sample_rate / rate_bpm[1])))
        maximum_lag = min(x.size - 1, int(np.ceil(60.0 * sample_rate / rate_bpm[0])))
    if maximum_lag <= minimum_lag:
        return _invalid("periodicity_autocorrelation", "aperiodicity", "window is too short")
    section = correlation[minimum_lag : maximum_lag + 1]
    relative_lag = int(np.argmax(section))
    peak = float(section[relative_lag])
    lag = minimum_lag + relative_lag
    quality = np.clip(max(peak, 0.0) * finite_fraction, 0.0, 1.0)
    return IndicatorResult(
        "periodicity_autocorrelation",
        peak,
        float(quality),
        "aperiodicity",
        {"lag_samples": lag, "rate_bpm": 60.0 * sample_rate / lag},
    )

    @dataclass(frozen=True)
class FusionModel:
    """Serializable logistic model followed by piecewise-linear isotonic calibration."""

    name: str
    features: tuple[str, ...]
    weights: tuple[float, ...]
    intercept: float
    calibration_x: tuple[float, ...] = (0.0, 1.0)
    calibration_y: tuple[float, ...] = (0.0, 1.0)
    missing_value: float = 0.5
    provenance: dict[str, Any] = field(default_factory=dict)

    def __post_init__(self) -> None:
        if not self.features or len(self.features) != len(self.weights):
            raise ValueError("features and weights must be non-empty and equally sized")
        if len(self.calibration_x) != len(self.calibration_y) or len(self.calibration_x) < 2:
            raise ValueError("calibration arrays must have equal length and at least two points")
        if any(right < left for left, right in pairwise(self.calibration_x)):
            raise ValueError("calibration_x must be nondecreasing")
        if any(right < left for left, right in pairwise(self.calibration_y)):
            raise ValueError("calibration_y must be nondecreasing")

    def predict(self, indicators: tuple[IndicatorResult, ...]) -> FusionOutput:
        """Fuse valid indicator qualities; missing features use the documented neutral value."""

        qualities = {
            indicator.name: indicator.quality
            for indicator in indicators
            if indicator.valid and indicator.quality is not None
        }
        contributions: dict[str, float] = {}
        logit = self.intercept
        for feature, weight in zip(self.features, self.weights, strict=True):
            value = float(qualities.get(feature, self.missing_value))
            contribution = weight * value
            contributions[feature] = contribution
            logit += contribution
        raw = _sigmoid(logit)
        calibrated = float(
            np.interp(
                raw,
                np.asarray(self.calibration_x),
                np.asarray(self.calibration_y),
                left=self.calibration_y[0],
                right=self.calibration_y[-1],
            )
        )
        return FusionOutput(raw, float(np.clip(calibrated, 0.0, 1.0)), contributions)

    def to_dict(self) -> dict[str, Any]:
        return {
            "name": self.name,
            "features": list(self.features),
            "weights": list(self.weights),
            "intercept": self.intercept,
            "calibration": {"x": list(self.calibration_x), "y": list(self.calibration_y)},
            "missing_value": self.missing_value,
            "provenance": self.provenance,
        }

    def save(self, path: str | Path) -> None:
        Path(path).write_text(json.dumps(self.to_dict(), indent=2) + "\n", encoding="utf-8")

    @classmethod
    def from_dict(cls, payload: dict[str, Any]) -> FusionModel:
        calibration = payload.get("calibration", {})
        return cls(
            name=str(payload["name"]),
            features=tuple(str(item) for item in payload["features"]),
            weights=tuple(float(item) for item in payload["weights"]),
            intercept=float(payload["intercept"]),
            calibration_x=tuple(float(item) for item in calibration.get("x", (0.0, 1.0))),
            calibration_y=tuple(float(item) for item in calibration.get("y", (0.0, 1.0))),
            missing_value=float(payload.get("missing_value", 0.5)),
            provenance=dict(payload.get("provenance", {})),
        )

    @classmethod
    def load(cls, path: str | Path) -> FusionModel:
        payload = json.loads(Path(path).read_text(encoding="utf-8"))
        if not isinstance(payload, dict):
            raise ValueError("model JSON root must be an object")
        return cls.from_dict(payload)

    def fit_fusion_model(
    features: npt.ArrayLike,
    labels: npt.ArrayLike,
    feature_names: tuple[str, ...],
    *,
    name: str = "fitted-logistic-isotonic",
    random_state: int = 0,
) -> FusionModel:
    """Fit logistic fusion with isotonic calibration on out-of-fold predictions."""

    x = np.asarray(features, dtype=np.float64)
    y = np.asarray(labels, dtype=np.int64).reshape(-1)
    if x.ndim != 2 or x.shape[1] != len(feature_names):
        raise ValueError("features must be a 2D array matching feature_names")
    if x.shape[0] != y.size or y.size < 4:
        raise ValueError("labels must match rows and contain at least four observations")
    if not np.all(np.isfinite(x)):
        raise ValueError("training features must be finite")
    unique = np.unique(y)
    if not np.array_equal(unique, np.array([0, 1])):
        raise ValueError("labels must contain both binary classes 0 and 1")
    class_counts = np.bincount(y, minlength=2)
    if int(class_counts.min()) < 2:
        raise ValueError("each class needs at least two observations for calibration")
    folds = min(5, int(class_counts.min()))
    splitter = StratifiedKFold(n_splits=folds, shuffle=True, random_state=random_state)
    logistic = LogisticRegression(max_iter=2000, class_weight="balanced", random_state=random_state)
    raw = cross_val_predict(
        logistic,
        x,
        y,
        cv=splitter,
        method="predict_proba",
    )[:, 1]
    calibrator = IsotonicRegression(out_of_bounds="clip")
    calibrator.fit(raw, y)
    logistic.fit(x, y)
    calibration_x = tuple(float(item) for item in calibrator.X_thresholds_)
    calibration_y = tuple(float(item) for item in calibrator.y_thresholds_)
    return FusionModel(
        name=name,
        features=feature_names,
        weights=tuple(float(item) for item in logistic.coef_[0]),
        intercept=float(logistic.intercept_[0]),
        calibration_x=calibration_x,
        calibration_y=calibration_y,
    )

    def attribute_artifacts(
    indicators: tuple[IndicatorResult, ...],
    model: FusionModel,
) -> dict[str, float]:
    """Aggregate ``1 - quality`` by artifact using absolute model weights.

    Values are relative diagnostic severities, not probabilities. Invalid or
    non-applicable indicators are omitted instead of being silently called bad.
    """

    weights = dict(zip(model.features, model.weights, strict=True))
    weighted_deficit: defaultdict[str, float] = defaultdict(float)
    total_weight: defaultdict[str, float] = defaultdict(float)
    for indicator in indicators:
        if not indicator.valid or indicator.quality is None:
            continue
        weight = abs(weights.get(indicator.name, 1.0))
        weighted_deficit[indicator.artifact] += weight * (1.0 - indicator.quality)
        total_weight[indicator.artifact] += weight
    severities = {
        artifact: float(np.clip(deficit / total_weight[artifact], 0.0, 1.0))
        for artifact, deficit in weighted_deficit.items()
        if total_weight[artifact] > 0
    }
    return dict(sorted(severities.items(), key=lambda item: (-item[1], item[0])))

    @dataclass(frozen=True)
class Recording:
    """A sampled, channel-major biomedical recording."""

    data: FloatArray
    sample_rate: float
    channel_names: tuple[str, ...] = ()
    signal_type: SignalType = SignalType.GENERIC
    source: str | None = None
    metadata: dict[str, Any] = field(default_factory=dict)

    def __post_init__(self) -> None:
        data = np.asarray(self.data, dtype=np.float64)
        if data.ndim == 1:
            data = data[np.newaxis, :]
        if data.ndim != 2:
            raise ValueError("recording data must be one- or two-dimensional")
        if data.shape[0] < 1:
            raise ValueError("recording must contain at least one channel")
        if not np.isfinite(self.sample_rate) or self.sample_rate <= 0:
            raise ValueError("sample_rate must be finite and positive")
        signal_type = (
            self.signal_type
            if isinstance(self.signal_type, SignalType)
            else SignalType(str(self.signal_type).lower())
        )
        names = self.channel_names or tuple(f"ch{i}" for i in range(data.shape[0]))
        if len(names) != data.shape[0]:
            raise ValueError("channel_names must match the channel count")
        object.__setattr__(self, "data", data)
        object.__setattr__(self, "channel_names", tuple(names))
        object.__setattr__(self, "signal_type", signal_type)

    @property
    def n_channels(self) -> int:
        return int(self.data.shape[0])

    @property
    def n_samples(self) -> int:
        return int(self.data.shape[1])

    @property
    def duration(self) -> float:
        return self.n_samples / self.sample_rate
